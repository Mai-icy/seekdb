cmake_minimum_required(VERSION 3.20)
project(seekdb_sirius_bridge LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Threads REQUIRED)
set(SEEKDB_SIRIUS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(SEEKDB_SIRIUS_API_DIR "${SEEKDB_SIRIUS_ROOT}/src/sql/engine/sirius")
add_library(sirius_client STATIC "${SEEKDB_SIRIUS_API_DIR}/sirius_client.cpp")
target_include_directories(sirius_client PUBLIC "${SEEKDB_SIRIUS_API_DIR}")
target_link_libraries(sirius_client PUBLIC ${CMAKE_DL_LIBS} Threads::Threads)
add_library(sirius_arrow_json STATIC arrow_json.cpp)
target_include_directories(sirius_arrow_json PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(sirius_execute_plan execute_plan.cpp)
target_link_libraries(sirius_execute_plan PRIVATE sirius_client)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_library(sirius_aggregate_client STATIC "${SEEKDB_SIRIUS_API_DIR}/sirius_aggregate_client.cpp")
  target_include_directories(sirius_aggregate_client PUBLIC "${SEEKDB_SIRIUS_API_DIR}")
  add_library(seekdb_sirius_proxy SHARED proxy.cpp)
  target_include_directories(seekdb_sirius_proxy PRIVATE "${SEEKDB_SIRIUS_API_DIR}")
  set_target_properties(seekdb_sirius_proxy PROPERTIES CXX_VISIBILITY_PRESET hidden)
endif()

option(SEEKDB_SIRIUS_BUILD_BRIDGE "Build the GPU bridge against an installed Sirius library" OFF)
option(SEEKDB_SIRIUS_ARROW_INPUT "Enable the pinned experimental Sirius Arrow-input overlay" OFF)
if(SEEKDB_SIRIUS_BUILD_BRIDGE)
  # Explicit paths also support an existing Sirius build tree without installing it.
  set(SIRIUS_INCLUDE_DIR "" CACHE PATH "Directory containing sirius/ffi.hpp")
  set(SIRIUS_LIBRARY "" CACHE FILEPATH "Path to libsirius.so")
  if(SIRIUS_INCLUDE_DIR AND SIRIUS_LIBRARY)
    add_library(seekdb_sirius_backend SHARED IMPORTED)
    set_target_properties(seekdb_sirius_backend PROPERTIES
      IMPORTED_LOCATION "${SIRIUS_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${SIRIUS_INCLUDE_DIR}")
  else()
    find_package(sirius CONFIG REQUIRED)
    add_library(seekdb_sirius_backend ALIAS sirius::sirius)
  endif()
  add_library(seekdb_sirius_bridge SHARED bridge.cpp)
  if(SEEKDB_SIRIUS_ARROW_INPUT)
    target_compile_definitions(seekdb_sirius_bridge PRIVATE SEEKDB_SIRIUS_ARROW_INPUT=1)
  endif()
  target_include_directories(seekdb_sirius_bridge PRIVATE "${SEEKDB_SIRIUS_API_DIR}")
  target_link_libraries(seekdb_sirius_bridge PRIVATE seekdb_sirius_backend sirius_arrow_json Threads::Threads)
  set_target_properties(seekdb_sirius_bridge PROPERTIES CXX_VISIBILITY_PRESET hidden)
  install(TARGETS seekdb_sirius_bridge LIBRARY DESTINATION lib)
endif()

include(CTest)
if(BUILD_TESTING)
  add_subdirectory("${SEEKDB_SIRIUS_ROOT}/unittest/sql/sirius" "sirius_tests")
endif()
